\chapter{Tìm hiểu và nghiên cứu thị trường}\label{chap:tim-hieu-thi-truong}

\section{Thực trạng marketing SME F\&B tại Việt Nam}\label{sec:thuc-trang-sme}

Thị trường F\&B tại Việt Nam có tính cạnh tranh rất cao, với sự bùng nổ của các nền tảng mạng xã hội như Facebook và TikTok đóng vai trò then chốt trong việc thu hút và giữ chân khách hàng. Xu hướng nội dung ngắn (short video, reel) và ``trending'' khiến tần suất đăng tải và độ bắt nhịp với văn hóa địa phương trở thành yếu tố cạnh tranh trực tiếp, trong khi đội ngũ vận hành quán thường không đủ thời gian để biến xu hướng thành kịch bản ổn định.

Tuy nhiên, các chủ doanh nghiệp F\&B vừa và nhỏ (SME) đang đối mặt với nhiều thách thức trong hoạt động marketing:

\begin{itemize}
  \item \textbf{Thiếu kiến thức và kỹ năng chuyên môn:} Hầu hết chủ quán là chuyên gia về ẩm thực và dịch vụ, nhưng lại thiếu kiến thức nền tảng về marketing số, sáng tạo nội dung và tối ưu hóa quảng cáo.
  \item \textbf{Hạn chế về thời gian và nguồn lực:} Lịch trình vận hành quán ăn bận rộn khiến họ không có đủ thời gian để duy trì sự nhất quán trong việc tạo và đăng tải nội dung hàng ngày.
  \item \textbf{Nội dung thiếu chuyên nghiệp và ngẫu hứng:} Các phương pháp hiện tại chủ yếu dựa vào chụp ảnh bằng điện thoại không qua chỉnh sửa chuyên nghiệp hoặc đăng bài theo cảm hứng tức thời, dẫn đến chất lượng nội dung thấp, không có chiến lược rõ ràng (thiếu khung hook--thân bài--CTA và thông điệp thương hiệu thống nhất).
  \item \textbf{Chi phí Agency đắt đỏ:} Việc thuê các agency marketing chuyên nghiệp thường vượt quá ngân sách của các SME, khiến họ phải tự xoay sở với các công cụ rời rạc (thiết kế, chat AI, lịch đăng).
\end{itemize}

Những vấn đề này dẫn đến việc các SME F\&B bỏ lỡ cơ hội tiếp cận khách hàng tiềm năng, giảm hiệu quả kinh doanh và gặp khó khăn trong việc xây dựng thương hiệu bền vững. Kết quả là nhiều SME hiểu tầm quan trọng của mạng xã hội nhưng \emph{thiếu khả năng thực thi nhất quán}. Do đó, nhu cầu về một giải pháp tự động hóa, thông minh, giá cả phải chăng, gần với thực tế vận hành (production-ready), phù hợp người dùng không chuyên công nghệ là rất cấp thiết.

\subsection{Recall}
\begin{itemize}
  \item \textbf{Đặc điểm:} Recall là một nền tảng AI-driven knowledge management, giúp người dùng tổ chức, lưu trữ và truy xuất thông tin từ nhiều nguồn dữ liệu phi cấu trúc như PDF, email, văn bản, ghi chú, blog, hoặc transcript video. Hệ thống sử dụng các mô hình ngôn ngữ lớn (LLM) kết hợp với vector search và retrieval-augmented generation (RAG) để hỗ trợ người dùng tìm kiếm thông tin một cách chính xác và nhanh chóng.
  \item \textbf{Phân tích:}
  \begin{itemize}
    \item \textit{Điểm mạnh:}
    \begin{itemize}
      \item Tự động hóa trích xuất tri thức: có khả năng nhận diện và trích xuất thông tin từ nhiều loại dữ liệu phi cấu trúc mà không cần người dùng tổ chức thủ công.
      \item Truy xuất thông tin chính xác: LLM kết hợp RAG và vector database giúp trả lời câu hỏi dựa trên ngữ cảnh, giảm sai lệch thông tin.
      \item Hỗ trợ hợp tác nhóm: dữ liệu cá nhân có thể được chia sẻ thành knowledge base chung.
      \item Khả năng học và mở rộng: hệ thống có thể mở rộng và học từ dữ liệu mới, tăng độ chính xác theo thời gian.
    \end{itemize}
    \item \textit{Điểm yếu:}
    \begin{itemize}
      \item Phụ thuộc vào chất lượng dữ liệu đầu vào: nếu dữ liệu phi cấu trúc kém chất lượng hoặc thiếu ngữ cảnh, kết quả truy xuất có thể không chính xác.
      \item Chi phí và phức tạp triển khai: so với các công cụ truyền thống như Notion, Recall yêu cầu hạ tầng AI và kiến thức cơ bản về mô hình ngôn ngữ để tối ưu hóa hệ thống.
      \item Yêu cầu làm quen: người dùng vẫn cần làm quen với cách hệ thống truy xuất thông tin dựa trên câu hỏi tự nhiên.
    \end{itemize}
  \end{itemize}
\end{itemize}

\section{Khảo sát nhu cầu người dùng}\label{sec:khao-sat}

\subsection{Mục đích và đối tượng}
Khảo sát thực tế được tiến hành nhằm:

\begin{itemize}
  \item Hiểu thực trạng hoạt động marketing của các chủ doanh nghiệp F\&B tại Việt Nam, bao gồm cách họ tìm kiếm cảm hứng, sản xuất nội dung, và quản lý các kênh mạng xã hội (Facebook, TikTok).
  \item Nhận diện các vấn đề và khó khăn mà chủ quán F\&B gặp phải trong việc duy trì sự nhất quán, chất lượng nội dung, và sự thiếu hụt kiến thức marketing chuyên môn.
  \item Thu thập ý kiến đóng góp từ thực tiễn về các giải pháp tự động hóa mong muốn: mức độ sẵn sàng sử dụng AI để biến cảm hứng (video/blog) thành bài đăng hoàn chỉnh, và mức độ tin cậy vào việc AI tự động đăng bài (sau duyệt).
  \item Nắm bắt thói quen sử dụng mạng xã hội, công cụ hiện có, hạn chế và mong muốn cải thiện hiệu quả marketing, làm cơ sở thiết kế hệ thống AI Agent thân thiện với người dùng không chuyên IT.
\end{itemize}

\textbf{Đối tượng khảo sát:}
\begin{itemize}
  \item Chủ sở hữu / Chủ quán (Owner): người trực tiếp điều hành, ra quyết định về chiến lược kinh doanh và marketing tại các quán ăn, nhà hàng, quán cà phê, tiệm bánh, v.v.
  \item Quản lý nhà hàng / Quản lý chi nhánh (Manager): người giám sát hoạt động hàng ngày, thường tham gia tạo nội dung marketing.
\end{itemize}

\textbf{Mẫu khảo sát tối thiểu (thiết kế):} mỗi ngành hàng tối thiểu 3 doanh nghiệp.

\subsection{Kế hoạch và nội dung phiếu khảo sát}
Kế hoạch khảo sát nhằm đảm bảo thu thập dữ liệu phản ánh đúng thực trạng SME. Nội dung phiếu gồm các phần chính sau (trích theo phiếu đã dùng trong đợt khảo sát):

\textbf{Phần 1 --- Thông tin cá nhân và doanh nghiệp} (xác định đối tượng và bối cảnh kinh doanh): họ tên; tên quán/công ty; vị trí công việc (chủ sở hữu/founder, quản lý nhà hàng/chi nhánh, nhân viên marketing nội bộ, khác); loại hình kinh doanh (nhà hàng tại chỗ, quán cà phê/đồ uống, tiệm bánh/dessert, online delivery/cloud kitchen, khác).

\textbf{Phần 2 --- Thực trạng marketing trên mạng xã hội:}
\begin{enumerate}
  \item Kênh đang dùng (chọn nhiều): Facebook Page, TikTok, Instagram, YouTube, Zalo Business, khác.
  \item Thời gian \emph{mỗi tuần} cho marketing MXH: dưới 1 giờ; 1--3 giờ; 3--5 giờ; hơn 5 giờ.
  \item Phương pháp chính để tạo ý tưởng/nội dung (thường dùng nhất): tự lướt MXH và lưu thủ công; thuê agency/freelancer; công cụ AI/thiết kế rời rạc (Canva, ChatGPT); không có phương pháp cụ thể, làm ngẫu hứng.
  \item Vấn đề lớn nhất (tối đa 2 lựa chọn): thiếu ý tưởng; không có thời gian duy trì đăng đều; chất lượng hình ảnh/video thấp; không biết viết nội dung hấp dẫn; không biết tối ưu theo xu hướng địa phương (hashtag, âm nhạc trend).
\end{enumerate}

\textbf{Phần 3 --- Nhu cầu giải pháp AI Marketing:} mức độ sẵn sàng dùng AI Agent biến video/bài viết cảm hứng thành lịch đăng hoàn chỉnh; hình thức giao diện ưa tiên (extension trình duyệt, mobile app, desktop app, chatbot Messenger/Zalo); mức độ thoải mái khi AI tự động đăng sau bước duyệt.

\textbf{Phần 4 --- Đánh giá mức độ cần thiết các tính năng} (thang 1--5): (A) thu thập cảm hứng từ trình duyệt; (B) AI phân tích cảm hứng thành ý tưởng marketing; (C) AI tạo caption/giọng văn chuyên nghiệp; (D) AI chỉnh sửa/nâng cao chất lượng ảnh/video đầu vào; (E) tự động lên lịch và đăng bài Facebook/TikTok; (F) chatbot trả lời theo nội dung đã lưu; (G) đề xuất xu hướng (hashtag, âm nhạc) tại Việt Nam.

\textbf{Phần 5 --- Đề xuất thêm:} tính năng mong muốn khác.

\subsection{Kết quả khảo sát}

\textbf{Số lượng mẫu:} 6 doanh nghiệp SME ngành Thực phẩm \& Đồ uống (F\&B) tại thị trường Việt Nam.

\textbf{Phân loại đối tượng:}
\begin{itemize}
  \item \textbf{Theo ngành hàng F\&B:} 83\% (5 mẫu) thuộc lĩnh vực quán cà phê/đồ uống quy mô nhỏ; 17\% (1 mẫu) nhà hàng ăn uống.
  \item \textbf{Theo vị trí công việc:} 83\% (5 mẫu) chủ sở hữu/chủ quán; 17\% (1 mẫu) quản lý nhà hàng.
\end{itemize}

\textbf{Tóm tắt các vấn đề cốt lõi thường gặp:}

Kết quả xác nhận các chủ quán F\&B quy mô nhỏ nhận thức rõ tầm quan trọng của marketing trên mạng xã hội (\textbf{100\%} sử dụng Facebook, đa số sử dụng thêm TikTok) nhưng gặp rào cản lớn khi triển khai hiệu quả.

\begin{itemize}
  \item \textbf{Thời gian và tính nhất quán:} ``Không có thời gian duy trì sự nhất quán (đăng bài đều đặn)'' được xem là rào cản lớn nhất. Đa số dành \textbf{dưới 3 giờ/tuần} cho hoạt động này, cho thấy nhu cầu tự động hóa.
  \item \textbf{Chất lượng và sáng tạo nội dung:} ``Thiếu ý tưởng nội dung sáng tạo'' và ``Chất lượng hình ảnh/video thấp'' phổ biến; nội dung thường ngẫu hứng hoặc lưu cảm hứng thủ công, thiếu chuyên nghiệp.
  \item \textbf{Giải pháp hiện tại:} công cụ rời rạc (Excel, ghi chú thủ công) hoặc phần mềm quản lý cơ bản, \emph{không} hỗ trợ trọn vẹn tạo nội dung hay tự động hóa đăng bài marketing.
\end{itemize}

\subsection{Gợi ý từ phản hồi định tính}
\begin{itemize}
  \item Ưu tiên phát triển trên nền tảng \textbf{ứng dụng di động} (chủ quán thường xuyên di chuyển) hoặc tích hợp kênh nhắn tin quen thuộc (\textbf{Zalo, Messenger}) khi đang vận hành tại quán.
  \item Yêu cầu bắt buộc có bước \textbf{duyệt (approval)} trước khi AI tự động đăng lên MXH, để kiểm soát rủi ro thương hiệu.
  \item Mong muốn AI gợi ý \textbf{thời điểm vàng (prime time)} đăng bài, tối ưu tương tác tại thị trường địa phương.
\end{itemize}

\subsection{Kết luận khảo sát}
Từ 6 doanh nghiệp SME F\&B (chủ yếu quán cà phê/đồ uống nhỏ và một nhà hàng), có thể rút ra:

Vấn đề cốt lõi không phải thiếu nhận thức mà thiếu khả năng thực thi: chủ quán hiểu tầm quan trọng của Facebook/TikTok, song rào cản lớn nhất là hạn chế thời gian và nguồn lực để duy trì đăng bài đều đặn.

Phương pháp thủ công kém hiệu quả: tìm cảm hứng ngẫu nhiên và lưu trữ thủ công hoặc qua công cụ rời rạc (Excel, ghi chú điện thoại) gây lãng phí thời gian và chất lượng nội dung không đồng nhất.

Nhu cầu tự động hóa cao: tính năng lên lịch và đăng bài được đánh giá quan trọng nhất (\textbf{4{,}8/5}), khẳng định nhu cầu giải phóng thời gian vận hành.

Cần cân bằng AI và kiểm soát con người: dù mong muốn tự động hóa, người được hỏi nhấn mạnh bước duyệt cuối cùng để an toàn thương hiệu.

Ưu tiên nội dung marketing hơn ``quản lý tri thức'' dạng chatbot: các tính năng sáng tạo nội dung (caption, phân tích cảm hứng, trend địa phương) được ưu tiên cao hơn chatbot truy xuất kho cũ (F: 3{,}5/5).

Kết quả khảo sát phù hợp với bối cảnh đặt vấn đề của luận văn; các hạn chế hiện tại làm giảm khả năng thu hút và giữ chân khách hàng. Do đó, phát triển AI Agent marketing chuyên biệt cho SME F\&B (thu thập cảm hứng, tạo nội dung, lên lịch đăng, có kiểm duyệt) là hướng giải pháp phù hợp và cấp thiết.

\section{Sự cần thiết của bước kiểm duyệt và hướng đánh giá tự động}\label{sec:approval-pipeline}

Kết quả khảo sát cho thấy mâu thuẫn vận hành quen thuộc: SME muốn \emph{tự động hóa cao} (điểm ưu tiên E đạt 4{,}8/5) để tiết kiệm thời gian, nhưng không chấp nhận rủi ro nội dung sai sót, nhạy cảm hoặc lệch giọng điệu thương hiệu; bước duyệt thủ công được yêu cầu rõ trong phản hồi định tính.

Hướng tiếp cận của đề tài là bổ sung \textbf{pipeline đánh giá đa lớp} trước khi đăng:

\begin{itemize}
  \item \textbf{Lớp quy tắc (rule-based):} chặn từ khóa nhạy cảm, kiểm tra thực thể bắt buộc (tên món, giá, địa chỉ), kiểm tra cấu trúc hook--thân--CTA.
  \item \textbf{Lớp xác suất:} dùng logprobs/perplexity làm tín hiệu trôi chảy hoặc bất thường ngôn ngữ, hỗ trợ lọc nhanh các bản nháp yếu.
  \item \textbf{Lớp định tính:} LLM-as-a-Judge (ví dụ G-Eval) hoặc persona khách hàng mô phỏng để chấm điểm đa tiêu chí.
\end{itemize}

Kết hợp ba lớp giúp giảm tải cho người duyệt: họ chỉ tập trung vào các bản đã vượt ngưỡng tự động hoặc đã được gắn cờ rủi ro, mở đường cho cơ chế \emph{auto-approval có kiểm soát} trong các bước sau của nghiên cứu (Chương 4--5).
