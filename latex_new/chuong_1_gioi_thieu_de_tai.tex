\chapter{Giới thiệu đề tài}\label{chap:gioi-thieu}

\section{Đặt vấn đề}\label{sec:dat-van-de}

Trong bối cảnh bùng nổ của mạng xã hội, đặc biệt là \textbf{Facebook} tại
thị trường Việt Nam, hoạt động marketing đóng vai trò sống còn đối với
các doanh nghiệp vừa và nhỏ (SME) ngành Thực phẩm \& Đồ uống (F\&B).
Tuy nhiên, các chủ doanh nghiệp F\&B thường là chuyên gia trong lĩnh vực
ẩm thực nhưng lại thiếu kiến thức, kỹ năng và thời gian để xây dựng,
triển khai một chiến lược marketing số nhất quán và hiệu quả. Họ gặp khó
khăn trong việc sáng tạo nội dung hấp dẫn, quản lý lịch đăng bài đa
kênh, và tương tác kịp thời với khách hàng.

Các giải pháp marketing agency truyền thống quá tốn kém, trong khi các
công cụ quản lý mạng xã hội hiện tại thường yêu cầu người dùng phải tự
tạo nội dung và quản lý workflow thủ công, gây tốn kém thời gian và
không giải quyết được vấn đề thiếu ý tưởng sáng tạo. Sự thiếu nhất quán
trong nội dung và quản lý chiến dịch dẫn đến việc bỏ lỡ cơ hội thu hút
khách hàng mới và giữ chân khách hàng hiện tại.

Trong bối cảnh đó, việc ứng dụng công nghệ AI Agent dựa trên các mô hình
ngôn ngữ lớn (LLM) mở ra một hướng giải quyết đột phá. Một AI Agent
chuyên biệt có thể hỗ trợ xây dựng chiến lược và tạo ra nội dung
marketing phù hợp thương hiệu từ dữ liệu nội bộ của doanh nghiệp, đồng
thời hỗ trợ đăng tải lên các kênh mạng xã hội (trong phạm vi đề tài,
trọng tâm là \textbf{Facebook Page} và định dạng nội dung trên Facebook).

Vì vậy, việc nghiên cứu và phát triển một AI Agent tự động hóa marketing
chuyên biệt cho SME ngành F\&B là vô cùng cần thiết và có ý nghĩa thực
tiễn. Đề tài này tập trung xây dựng hệ thống không chỉ giúp chủ doanh
nghiệp tiết kiệm thời gian, tối ưu hóa nguồn lực mà còn nâng cao hiệu
quả chiến dịch marketing thông qua kỹ thuật tinh chỉnh mô hình (QLoRA,
GRPO) và cơ chế đánh giá chất lượng nội dung đa lớp trước khi đăng bài.

\section{Mục tiêu nghiên cứu}\label{sec:muc-tieu}

Đề tài hướng tới phát triển một AI Agent nhằm hỗ trợ các chủ doanh
nghiệp F\&B tự động hóa quy trình tạo nội dung marketing. Hệ thống giúp
người dùng không chuyên về marketing chuyển đổi dữ liệu thương hiệu và
sản phẩm thành các nội dung chuyên nghiệp, bao gồm caption, kịch bản bài
đăng và kế hoạch nội dung có cấu trúc, dưới dạng sẵn sàng để xem xét và
đăng tải lên mạng xã hội.

Các mục tiêu cụ thể của hệ thống bao gồm:

\begin{itemize}
\item
  \textbf{Xây dựng Agent chuyên biệt:} Xây dựng một tác nhân (Agent) có
  kiến trúc phù hợp với bối cảnh marketing F\&B trên Facebook, có khả
  năng tận dụng tri thức thương hiệu và dữ liệu nội bộ của doanh nghiệp
  thay vì chỉ dựa trên LLM tổng quát.

\item
  \textbf{Nghiên cứu kỹ thuật tinh chỉnh mô hình:} Áp dụng
  hai phương pháp tinh chỉnh chính là học có giám sát (Supervised Fine-Tuning) và học tăng cường (Reinforcement Learning) nhằm định hình phong cách
  marketing và hành vi sinh nội dung phù hợp với đặc thù ngành F\&B và
  thị trường Việt Nam.

\item
  \textbf{Thiết lập pipeline đánh giá chất lượng nội dung:} Xây dựng cơ
  chế kiểm tra và chấm điểm đa lớp kết hợp quy tắc cứng, chỉ số xác
  suất và đánh giá định tính (LLM-as-a-Judge, G-Eval) để hỗ trợ quyết
  định đăng bài, giảm rủi ro nội dung kém chất lượng hoặc không phù hợp
  thương hiệu.

\item
  \textbf{Quản lý và cung cấp nội dung:} Cho phép người dùng lưu trữ dữ
  liệu thương hiệu, xem xét, chỉnh sửa và tổ chức các nội dung đã được
  tạo ra theo chiến dịch hoặc thời gian cụ thể.

\item
  \textbf{Đánh giá hệ thống:} Đề xuất phương pháp kiểm định tương quan
  giữa pipeline đánh giá tự động và đánh giá của con người (Pearson,
  Spearman), từ đó xác định ngưỡng phê duyệt phù hợp và hướng cải thiện
  cho các giai đoạn tiếp theo.
\end{itemize}

\section{Đối tượng và phạm vi nghiên cứu}\label{sec:doi-tuong-pham-vi}

\textbf{Đối tượng:} Doanh nghiệp SME ngành F\&B tại Việt Nam; người dùng
mục tiêu là chủ quán, người quản lý hoặc nhân sự phụ trách marketing.

\textbf{Phạm vi nội dung và kênh:} Đề tài \textbf{tập trung vào nội dung
marketing trên Facebook} (Facebook Page, bài đăng và hình ảnh). Các nền tảng khác không thuộc phạm vi đề tài
nhằm thống nhất đánh giá và triển khai thực nghiệm.

\textbf{Phạm vi kỹ thuật:} Mô hình ngôn ngữ mã nguồn mở; tinh chỉnh
(SFT/RL); cơ chế truy xuất tri thức (RAG) phục vụ truy xuất thông tin thương hiệu; pipeline đánh giá chất lượng nội dung.

\section{Phương pháp nghiên cứu}\label{sec:phuong-phap}

Phương pháp nghiên cứu gồm ba nhóm hoạt động chính:

\begin{enumerate}
\item
  \textbf{Thực nghiệm so sánh (benchmarking):} So sánh các cấu hình mô
  hình và chiến lược sinh--đánh giá nội dung trên bộ kịch bản và chỉ số
  thống nhất, nhằm xác định phương án phù hợp nhất với SME F\&B.

\item
  \textbf{Tinh chỉnh mô hình mã nguồn mở:} Huấn luyện bổ sung trên dữ
  liệu và mục tiêu của đề tài (fine-tuning với SFT/RL), sử dụng mô
  hình mã nguồn mở có thể tái hiện được trong nghiên cứu.

\item
  \textbf{Kiểm định chất lượng:} Đánh giá độ tin cậy và chất lượng đầu
  ra của hệ thống thông qua pipeline đánh giá đa lớp: kết hợp quy tắc cứng và đánh giá định tính (LLM-as-a-Judge, G-Eval).
\end{enumerate}

\section{Thách thức}\label{sec:thach-thuc}

\begin{itemize}
\item
  \textbf{Thiếu kỹ năng và thời gian của chủ quán F\&B:}
  Chủ quán thường không chuyên về marketing và công nghệ, yêu cầu giao
  diện cực kỳ đơn giản và quy trình tự động hóa cao độ, giảm thiểu tối
  đa thao tác thủ công hàng ngày.

\item
  \textbf{Chất lượng nội dung và tính xác thực:}
  Nội dung marketing F\&B đòi hỏi sự chân thực và nhất quán với giọng
  thương hiệu, trong khi nội dung do AI tạo sinh có nguy cơ bị cảm nhận
  là thiếu tự nhiên hoặc không phù hợp.

\item
  \textbf{Tính địa phương hóa:}
  Xu hướng F\&B và nội dung viral trên \textbf{Facebook} tại thị trường
  Việt Nam thay đổi liên tục; hệ thống phải tạo ra nội dung phù hợp văn
  hóa địa phương, tránh biểu đạt lỗi thời hoặc không liên quan.
\end{itemize}

\section{Bố cục luận văn}\label{sec:bo-cuc}

Đề tài được trình bày trong năm chương:

\begin{itemize}
\item
  \textbf{Chương 1} giới thiệu vấn đề, mục tiêu, phạm vi và phương pháp
  nghiên cứu.
\item
  \textbf{Chương 2} trình bày thực trạng thị trường, phân tích các giải
  pháp hiện có và khảo sát nhu cầu của nhóm người dùng mục tiêu.
\item
  \textbf{Chương 3} trình bày cơ sở lý thuyết và công nghệ nền tảng: kiến
  trúc mô hình, fine-tuning, RAG và các chỉ số đánh giá.
\item
  \textbf{Chương 4} trình bày thiết kế hệ thống, pipeline đánh giá, kết
  quả thực nghiệm fine-tuning và phân tích kết quả.
\item
  \textbf{Chương 5} kết luận, đánh giá đóng góp, hạn chế và hướng phát
  triển tiếp theo.
\end{itemize}

Chương 1 đã xác lập bối cảnh, mục tiêu và phạm vi nghiên cứu làm nền
tảng cho toàn bộ luận văn. Các chương tiếp theo sẽ lần lượt trình bày
phân tích thị trường, cơ sở lý thuyết, hiện thực hóa hệ thống và kết
quả thực nghiệm.
