    \begin{column}{0.56\textwidth}
      \scriptsize
      \textbf{Tại sao 1 neuron sai về lý thuyết với GRPO?}
      \vspace{0.3em}
      \begin{enumerate}\setlength\itemsep{4pt}
        \item \textbf{Variance collapse:}
          $r_i \approx \text{const}$ trong nhóm $G$
          $\Rightarrow \sigma_r \approx 0
          \Rightarrow A_i = \frac{r_i-\bar{r}}{\sigma_r+\varepsilon} \approx 0$
          --- mất tín hiệu học.

        \item \textbf{1 scalar $\not\Leftrightarrow$ (F, E, V):}
          Không phân biệt được ``Faithfulness cao, Vibe thấp'' với ngược lại
          $\Rightarrow$ policy không biết chiều nào cần tối ưu.

        \item \textbf{Reward hacking:}
          Model tìm shortcut maximize $r$ bằng thêm emoji và CTA máy móc,
          bỏ qua bảo toàn thực thể.

        \item \textbf{Out-of-distribution:}
          Scorer trained trên review tiêu dùng, marketing content nằm ngoài
          phân phối $\Rightarrow r_i$ không tin cậy.
      \end{enumerate}
    \end{column}
  \end{columns}

    Phân tích thiết kế đúng $\downarrow$.
