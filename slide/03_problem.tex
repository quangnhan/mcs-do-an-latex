\begin{frame}[t]{Phát biểu bài toán}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.45\textwidth}
      \textbf{Đầu vào} $x = (b, s, m)$
      \begin{itemize}
        \item $b$: campaign brief, title
        \item $s$: seed content (giá, combo, thông số)
        \item $m$: metadata thương hiệu, ràng buộc Facebook
      \end{itemize}

      \textbf{Đầu ra} $y = (plan, caption, img)$
      \begin{itemize}
        \item Kế hoạch nội dung, caption F\&B
        \item Ảnh minh họa và metadata đăng bài
      \end{itemize}
      \begin{block}{Mục tiêu}
        $\max_\theta \mathbb{E}_x[\text{Overall}(\pi_\theta(x))]$,
        $\text{Overall}=\tfrac{1}{3}(F+E+V)$.
      \end{block}
    \end{column}
    \begin{column}{0.52\textwidth}
      \centering
      \begin{tikzpicture}[node distance=0.7cm, every node/.style={font=\small}]
        \node[draw, rounded corners, fill=softblue, text width=3.6cm, minimum height=0.9cm, align=center] (input) {Campaign brief $x$};
        \node[draw, rounded corners, fill=softgreen, below=of input, text width=3.6cm, minimum height=0.9cm, align=center] (agent) {Marketing Agent $\pi_\theta$};
        \node[draw, rounded corners, fill=softorange, below=of agent, text width=3.6cm, minimum height=0.9cm, align=center] (output) {Post candidate $y$};
        \node[draw, rounded corners, fill=softred, below=of output, text width=3.6cm, minimum height=0.9cm, align=center] (constraints) {$F,E,V$ $\Rightarrow$ Overall};
        \draw[-{Stealth}, thick] (input) -- (agent);
        \draw[-{Stealth}, thick] (agent) -- (output);
        \draw[-{Stealth}, thick] (constraints) -- (output);
      \end{tikzpicture}

      \begin{alertblock}{Ràng buộc}
        \footnotesize
        $\pi_\theta: x \mapsto y$;\;
        $\text{Overall}(y)=\tfrac{F+E+V}{3}\geq\theta$.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}
