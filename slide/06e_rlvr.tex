\begin{frame}[t]{RLVR: học từ tín hiệu reward tự động, không cần người gán nhãn}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.50\textwidth}
      \begin{block}{RLVR là gì?}
        \small
        \begin{itemize}
          \item Thay người gán nhãn bằng hàm reward tự động
          \item Phù hợp khi có tiêu chí rõ ràng: toán (đúng/sai), code (chạy được không)
          \item Mở rộng dữ liệu huấn luyện mà không cần người chấm
        \end{itemize}
      \end{block}
      \begin{block}{Khác RLHF ở điểm nào?}
        \small
        \begin{itemize}
          \item RLHF: người so sánh -- chậm, tốn kém
          \item RLVR: hàm tự động -- nhanh, rẻ hơn
          \item Đánh đổi: cần tiêu chí đánh giá định lượng được
        \end{itemize}
      \end{block}
    \end{column}
