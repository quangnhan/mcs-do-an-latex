\begin{frame}[t]{DPO: bỏ reward model, tối ưu trực tiếp từ cặp sở thích}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.50\textwidth}
      \begin{block}{DPO giải quyết thế nào?}
        \small
        \begin{itemize}
          \item Không cần huấn luyện reward model riêng
          \item Tối ưu trực tiếp từ cặp $(x,\,y^+,\,y^-)$
          \item Đơn giản hơn, ổn định hơn PPO
        \end{itemize}
      \end{block}
      \begin{block}{Yêu cầu dữ liệu}
        \small
        \begin{itemize}
          \item $y^+$ và $y^-$ phải sinh từ \textbf{cùng một prompt} $x$
          \item Cặp so sánh do người dùng hoặc LLM chấm điểm
        \end{itemize}
      \end{block}
    \end{column}
