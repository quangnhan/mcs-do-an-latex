\begin{frame}[t]{Lịch sử RL trong LLM: từ RLHF đến RLVR}
  \vspace{0.3cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      \begin{block}{RLHF -- Học từ phản hồi người dùng}
        \small
        \begin{itemize}
          \item Người dùng so sánh hai bài, chọn bài tốt hơn
          \item Tạo dữ liệu sở thích: bài tốt $y^+$ vs.\ bài kém $y^-$
          \item Model học ưu tiên $y^+$, tránh $y^-$
          \item Thuật toán nổi bật: \textbf{DPO}
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{RLVR -- Học từ tín hiệu có thể kiểm chứng}
        \small
        \begin{itemize}
          \item Thay người dùng bằng hàm reward tự động
          \item Phù hợp bài toán có đáp án rõ ràng (toán, code)
          \item Không cần gán nhãn thủ công
          \item Thuật toán nổi bật: \textbf{GRPO}
          \item \textit{\color{bkblue}Điều kiện quan trọng:} reward function phải \textbf{verifiable} và \textbf{đúng miền} --- nếu không, tín hiệu học bị nhiễu
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
