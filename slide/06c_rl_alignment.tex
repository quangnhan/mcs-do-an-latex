\begin{frame}[t]{RL Alignment: căn chỉnh model theo phản hồi người dùng}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Vấn đề SFT chưa giải quyết được}
        \small
        \begin{itemize}
          \item SFT chỉ dạy model viết \textit{giống} mẫu, chưa dạy model biết bài nào \textit{tốt hơn}
          \item Cần thêm tín hiệu: ``bài A hay hơn bài B'' từ cùng một prompt
        \end{itemize}
      \end{block}
      \begin{block}{Cách hoạt động}
        \small
        \begin{itemize}
          \item Sinh nhiều bài từ cùng 1 prompt
          \item Người dùng (hoặc bộ tiêu chí F/E/V) chọn bài tốt hơn
          \item Tạo cặp dữ liệu sở thích: bài tốt vs.\ bài kém
          \item Model học ưu tiên bài tốt, tránh bài kém
        \end{itemize}
      \end{block}
    \end{column}
